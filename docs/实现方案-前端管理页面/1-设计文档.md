# 实现方案：前端管理页面完善

## 1. 概述

### 问题分析
- **现状**: 前端 Angular 页面仅有基础的 CRUD，缺少高级功能（搜索、筛选、排序、批量操作）
- **风险**: 用户体验不佳，管理效率低，无法快速定位数据
- **目标**: 实现生产级管理界面，包括高级搜索、数据表格、详情展开、实时监控等

### 核心目标
- ✅ 高级搜索与筛选（多条件组合）
- ✅ 表格排序和分页（服务端）
- ✅ 批量操作（删除、修改状态等）
- ✅ 详情展开面板
- ✅ 快速编辑行内修改
- ✅ 实时对话和监控
- ✅ 导出功能（Excel/CSV）
- ✅ 权限控制

---

## 2. 技术栈

- **框架**: Angular 21+ (Standalone Components, Signals)
- **UI**: Angular Material v21
- **状态管理**: Signals（reactive primitives）
- **HTTP**: HttpClient + Interceptors
- **表格**: mat-table（虚拟滚动）
- **分页**: mat-paginator
- **实时通知**: SignalR
- **导出**: XLSX(write Excel), csv-parser

```typescript
export interface TableQuery<T extends Record<string, any> = Record<string, any>> {

  pageSize: number;

  /// 日期范围头尾

    start: Date;
    end: Date;
  };
}

export interface TableResponse<T> {
  items: T[];
  totalCount: number;
  pageIndex: number;
  pageSize: number;
  totalPages: number;

export interface TableColumn {
  name: string;
  label: string;
  width?: string;
  sortable?: boolean;
  filterable?: boolean;
  type?: 'text' | 'number' | 'date' | 'badge' | 'action';
  format?: (value: any) => string;
}
```

## 5. 核心服务实现

### 5.1 表格数据服务

```typescript
// shared/services/table-data.service.ts
import { Injectable, signal, computed } from '@angular/core';
import { HttpClient, HttpParams } from '@angular/common/http';
import { BehaviorSubject, Observable, debounceTime, distinctUntilChanged, switchMap } from 'rxjs';

@Injectable({ providedIn: 'root' })
export class TableDataService<T> {
  private readonly query$ = new BehaviorSubject<TableQuery>(this.getDefaultQuery());
  
  /// 表格数据
  private readonly data$ = new BehaviorSubject<TableResponse<T>>({
    items: [],
    totalCount: 0,
    pageIndex: 1,
    pageSize: 20,
    totalPages: 0
  });

  /// 加载中状态
  private readonly loading$ = new BehaviorSubject(false);
  
  /// 选中的行
  private readonly selectedRows = signal<Set<string>>(new Set());

  readonly data = computed(() => this.data$.getValue());
  readonly loading = computed(() => this.loading$.getValue());
  readonly selectedCount = computed(() => this.selectedRows().size);
  readonly isAllSelected = computed(() => 
    this.data().items.length > 0 && 
    this.selectedCount() === this.data().items.length
  );

  constructor(private http: HttpClient, private apiUrl: string) {
    this.setupDataFetching();
  }

  private setupDataFetching() {
    this.query$
      .pipe(
        debounceTime(300),
        distinctUntilChanged(),
        switchMap(query => this.fetchData(query))
      )
      .subscribe(response => {
        this.data$.next(response);
        this.loading$.next(false);
      });
  }

  private fetchData(query: TableQuery): Observable<TableResponse<T>> {
    this.loading$.next(true);
    
    let params = new HttpParams()
      .set('pageIndex', query.pageIndex.toString())
      .set('pageSize', query.pageSize.toString());

    if (query.sortBy) {
      params = params.set('sortBy', query.sortBy);
    }
    if (query.sortOrder) {
      params = params.set('sortOrder', query.sortOrder);
    }
    if (query.keyword) {
      params = params.set('keyword', query.keyword);
    }
    if (query.filters) {
      params = params.set('filtersJson', JSON.stringify(query.filters));
    }

    return this.http.get<TableResponse<T>>(`${this.apiUrl}/list`, { params });
  }

  updateQuery(updates: Partial<TableQuery>) {
    const current = this.query$.getValue();
    this.query$.next({ ...current, ...updates, pageIndex: 1 }); // 重置分页
  }

  setPageIndex(pageIndex: number) {
    this.updateQuery({ pageIndex });
  }

  setPageSize(pageSize: number) {
    this.updateQuery({ pageSize, pageIndex: 1 });
  }

  setSorting(sortBy: string, sortOrder: 'asc' | 'desc') {
    this.updateQuery({ sortBy, sortOrder, pageIndex: 1 });
  }

  search(keyword: string) {
    this.updateQuery({ keyword });
  }

  setFilters(filters: Record<string, any>) {
    this.updateQuery({ filters });
  }

  toggleRowSelection(id: string) {
    const selected = new Set(this.selectedRows());
    if (selected.has(id)) {
      selected.delete(id);
    } else {
      selected.add(id);
    }
    this.selectedRows.set(selected);
  }

  toggleSelectAll() {
    if (this.isAllSelected()) {
      this.selectedRows.set(new Set());
    } else {
      const ids = this.data().items.map(item => (item as any).id);
      this.selectedRows.set(new Set(ids));
    }
  }

  getSelectedIds(): string[] {
    return Array.from(this.selectedRows());
  }

  clearSelection() {
    this.selectedRows.set(new Set());
  }

  refresh() {
    this.updateQuery({ ...this.query$.getValue() });
  }
}
```

### 5.2 搜索筛选服务

```typescript
// shared/services/search-filter.service.ts
import { Injectable, signal } from '@angular/core';

export interface FilterConfig {
  field: string;
  label: string;
  type: 'text' | 'select' | 'date' | 'daterange' | 'number' | 'multiselect';
  placeholder?: string;
  options?: { label: string; value: any }[];
  defaultValue?: any;
}

@Injectable({ providedIn: 'root' })
export class SearchFilterService {
  private readonly filterConfigs = signal<FilterConfig[]>([]);
  private readonly activeFilters = signal<Record<string, any>>({});

  readonly filters = this.activeFilters.asReadonly();

  setFilterConfigs(configs: FilterConfig[]) {
    this.filterConfigs.set(configs);
  }

  updateFilter(field: string, value: any) {
    const current = this.activeFilters();
    if (value === null || value === undefined || value === '') {
      const updated = { ...current };
      delete updated[field];
      this.activeFilters.set(updated);
    } else {
      this.activeFilters.set({ ...current, [field]: value });
    }
  }

  clearFilters() {
    this.activeFilters.set({});
  }

  getFilterDisplayText(field: string, value: any): string {
    const config = this.filterConfigs().find(c => c.field === field);
    if (!config) return value?.toString() ?? '';

    if (config.type === 'select' || config.type === 'multiselect') {
      const option = config.options?.find(o => o.value === value);
      return option?.label ?? '';
    }

    if (config.type === 'date') {
      return new Date(value).toLocaleDateString();
    }

    return value?.toString() ?? '';
  }
}
```

### 5.3 导出服务

```typescript
// shared/services/export.service.ts
import { Injectable } from '@angular/core';
import * as XLSX from 'xlsx';

@Injectable({ providedIn: 'root' })
export class ExportService {
  
  exportToExcel<T>(data: T[], columns: { field: string; header: string }[], filename: string) {
    const ws = XLSX.utils.json_to_sheet(data, { header: columns.map(c => c.field) });
    
    // 设置表头
    columns.forEach((col, index) => {
      const cell = XLSX.utils.encode_cell({ r: 0, c: index });
      ws[cell].v = col.header;
    });

    const wb = XLSX.utils.book_new();
    XLSX.utils.book_append_sheet(wb, ws, 'Data');
    XLSX.writeFile(wb, `${filename}-${this.getTimestamp()}.xlsx`);
  }

  exportToCsv<T>(data: T[], columns: { field: string; header: string }[], filename: string) {
    const headers = columns.map(c => c.header).join(',');
    const rows = data.map(item => 
      columns.map(col => this.escapeCSV((item as any)[col.field])).join(',')
    );
    
    const csv = [headers, ...rows].join('\n');
    const blob = new Blob([csv], { type: 'text/csv' });
    const url = window.URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `${filename}-${this.getTimestamp()}.csv`;
    a.click();
  }

  private escapeCSV(value: any): string {
    if (value === null || value === undefined) return '';
    const str = value.toString();
    return str.includes(',') || str.includes('"') 
      ? `"${str.replace(/"/g, '""')}"` 
      : str;
  }

  private getTimestamp(): string {
    return new Date().toISOString().split('T')[0];
  }
}
```

### 7.1 后端列表 API

```csharp
[HttpGet("list")]
public async Task<ActionResult<TableResponseDto<AgentDto>>> ListAsync(
    [FromQuery] int pageIndex = 1,
    [FromQuery] int pageSize = 20,
    [FromQuery] string? keyword = null,
    [FromQuery] string? sortBy = "createdAt",
    [FromQuery] string? sortOrder = "desc",
    [FromQuery] string? filtersJson = null,
    CancellationToken cancellationToken = default
)
{
    var query = _context.Agents.AsQueryable();

    // 关键词搜索
    if (!string.IsNullOrEmpty(keyword))
    {
        query = query.Where(a => a.Name.Contains(keyword) || a.Description.Contains(keyword));
    }

    // 高级筛选
    if (!string.IsNullOrEmpty(filtersJson))
    {
        var filters = JsonSerializer.Deserialize<Dictionary<string, JsonElement>>(filtersJson);
        if (filters != null)
        {
            foreach (var filter in filters)
            {
                query = ApplyFilter(query, filter.Key, filter.Value);
            }
        }
    }

    // 排序
    query = sortBy?.ToLower() switch
    {
        "name" => sortOrder == "desc" ? query.OrderByDescending(a => a.Name) : query.OrderBy(a => a.Name),
        "status" => sortOrder == "desc" ? query.OrderByDescending(a => a.Status) : query.OrderBy(a => a.Status),
        _ => sortOrder == "desc" ? query.OrderByDescending(a => a.CreatedAt) : query.OrderBy(a => a.CreatedAt)
    };

    var totalCount = await query.CountAsync(cancellationToken);
    var items = await query
        .Skip((pageIndex - 1) * pageSize)
        .Take(pageSize)
        .Select(a => new AgentDto { /* 字段映射 */ })
        .ToListAsync(cancellationToken);

    return Ok(new TableResponseDto<AgentDto>
    {
        Items = items,
        TotalCount = totalCount,
        PageIndex = pageIndex,
        PageSize = pageSize
    });
}

private IQueryable<Agent> ApplyFilter(IQueryable<Agent> query, string field, JsonElement value)
{
    return field.ToLower() switch
    {
        "status" => query.Where(a => a.Status == value.GetString()),
        "createdAt" => value.TryGetProperty("start", out var start) && value.TryGetProperty("end", out var end)
            ? query.Where(a => a.CreatedAt >= start.GetDateTime() && a.CreatedAt <= end.GetDateTime())
            : query,
        _ => query
    };
}
```

```typescript
// shared/interceptors/auth.interceptor.ts
@Injectable()
export class AuthInterceptor implements HttpInterceptor {
  intercept(req: HttpRequest<any>, next: HttpHandler): Observable<HttpEvent<any>> {
    const token = localStorage.getItem('token');
    if (token) {
      req = req.clone({
        setHeaders: { Authorization: `Bearer ${token}` }
      });
    }
    return next.handle(req);
  }
}
```
