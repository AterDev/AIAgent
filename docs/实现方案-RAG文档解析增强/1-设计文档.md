# 实现方案：RAG 文档解析功能增强

## 1. 概述

### 问题分析
- **现状**: 仅支持纯文本和简单 Markdown；PDF、DOCX、XLSX 等常见格式无法支持
- **影响**: 用户无法上传实际业务文档（合同、报告、表格等），严重限制了 RAG 功能
- **目标**: 支持主流文档格式, 构建 **生产级** 文档解析能力

### 支持格式优先级

| 格式 | 优先级 | 难度 | 库 | 预计用量 |
|------|--------|------|----|----|
| PDF | 🔴 P0 | 中 | iTextSharp | 10-20% |
| MS Word (.docx) | 🔴 P0 | 低 | DocumentFormat.OpenXml | 10-15% |
| Excel (.xlsx) | 🟡 P1 | 低 | EPPlus | 5-10% |
| PowerPoint (.pptx) | 🟡 P1 | 中 | DocumentFormat.OpenXml | 5-10% |
| 扫描件 (含 OCR) | 🟡 P1 | 高 | Azure Computer Vision | 10-15% |
| JSON/XML | 🟡 P1 | 低 | System.Text.Json | 2-5% |

## 2. 架构设计

### 2.1 分层架构

```
┌─────────────────────────────────────────┐
│         RagDocumentService              │  对外接口
│   (文档上传、处理、状态管理)             │
└──────────┬──────────────────────────────┘
           │
┌──────────▼──────────────────────────────┐
│      DocumentParserFactory              │  解析工厂
│   (根据文件类型选择最优解析器)           │
└──────────┬──────────────────────────────┘
           │
    ┌──────┴──────┬──────────┬──────────┬──────────┐
    │             │          │          │          │
┌───▼────┐  ┌───▼────┐ ┌───▼────┐ ┌───▼────┐ ┌───▼────┐
│ Text   │  │ PDF    │ │ Word   │ │ Excel  │ │  PPT   │
│ Parser │  │ Parser │ │ Parser │ │ Parser │ │ Parser │
└────────┘  └────────┘ └────────┘ └────────┘ └────────┘
```

### 2.2 解析流程

```
文档上传
  ↓
文件类型识别 (MIME type + 扩展名)
  ↓
选择对应解析器 (DocumentParserFactory)
  ↓
解析文件 (IDocumentParser.ParseAsync)
  ↓
文本清理和规范化 (TextNormalizer)
  ↓
分块处理 (ITextChunker)
  ↓
向量化 (EmbeddingGenerator)
  ↓
存储 Qdrant
```

### 5.1 文档解析工厂

```csharp
namespace KnowledgeBaseMod.Services;

/// <summary>
/// 文档解析器工厂
/// </summary>
public interface IDocumentParserFactory
{
    /// <summary>
    /// 获取对应的文档解析器
    /// </summary>
    IDocumentParser GetParser(string contentType, string? fileName = null);
    
    /// <summary>
    /// 检查是否支持该类型
    /// </summary>
    bool IsSupported(string contentType);
}

public class DocumentParserFactory(
    ILogger<DocumentParserFactory> logger,
    PdfDocumentParser pdfParser,
    WordDocumentParser wordParser,
    ExcelDocumentParser excelParser,
    PowerPointDocumentParser pptParser,
    SimpleDocumentParser simpleParser
) : IDocumentParserFactory
{
    private readonly Dictionary<string, IDocumentParser> _parserMap = new(StringComparer.OrdinalIgnoreCase)
    {
        // PDF
        { "application/pdf", pdfParser },
        { ".pdf", pdfParser },

        // Word
        { "application/vnd.openxmlformats-officedocument.wordprocessingml.document", wordParser },
        { "application/msword", wordParser },
        { ".docx", wordParser },
        { ".doc", wordParser },

        // Excel
        { "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet", excelParser },
        { "application/vnd.ms-excel", excelParser },
        { ".xlsx", excelParser },
        { ".xls", excelParser },

        // PowerPoint
        { "application/vnd.openxmlformats-officedocument.presentationml.presentation", pptParser },
        { "application/vnd.ms-powerpoint", pptParser },
        { ".pptx", pptParser },
        { ".ppt", pptParser },

        // 文本
        { "text/plain", simpleParser },
        { "text/markdown", simpleParser },
        { "application/json", simpleParser },
        { "application/xml", simpleParser },
        { ".txt", simpleParser },
        { ".md", simpleParser },
        { ".json", simpleParser },
        { ".xml", simpleParser },
    };

    public IDocumentParser GetParser(string contentType, string? fileName = null)
    {
        // 优先使用 fileName 扩展名
        if (!string.IsNullOrWhiteSpace(fileName))
        {
            var ext = Path.GetExtension(fileName).ToLowerInvariant();
            if (_parserMap.TryGetValue(ext, out var parser))
            {
                return parser;
            }
        }

        // 使用 ContentType
        if (_parserMap.TryGetValue(contentType, out var parserByType))
        {
            return parserByType;
        }

        logger.LogWarning("Unsupported content type: {ContentType}, falling back to simple parser", contentType);
        return simpleParser;
    }

    public bool IsSupported(string contentType)
    {
        return _parserMap.ContainsKey(contentType) || _parserMap.ContainsKey(Path.GetExtension(contentType).ToLowerInvariant());
    }
}
```

### 5.2 PDF 解析器实现

```csharp
namespace KnowledgeBaseMod.Services;

using iText.Kernel.Pdf;
using iText.Kernel.Pdf.Canvas.Parser;
using iText.Kernel.Pdf.Canvas.Parser.Listener;

/// <summary>
/// PDF 文档解析器
/// </summary>
public class PdfDocumentParser(
    ILogger<PdfDocumentParser> logger
) : IDocumentParser
{
    public async Task<DocumentParseResult> ParseAsync(
        RagDocument document,
        string? rawContent,
        CancellationToken cancellationToken = default
    )
    {
        if (string.IsNullOrWhiteSpace(rawContent) && string.IsNullOrWhiteSpace(document.SourceUrl))
        {
            throw new BusinessException("PDF content must be provided via rawContent or SourceUrl");
        }

        byte[] pdfBytes;

        if (!string.IsNullOrWhiteSpace(rawContent))
        {
            // 如果 rawContent 是 base64 编码的
            try
            {
                pdfBytes = Convert.FromBase64String(rawContent);
            }
            catch
            {
                // 如果不是 base64，假设是二进制数据
                pdfBytes = Encoding.UTF8.GetBytes(rawContent);
            }
        }
        else if (!string.IsNullOrWhiteSpace(document.SourceUrl))
        {
            // 从 URL 下载
            using var client = new HttpClient();
            pdfBytes = await client.GetByteArrayAsync(document.SourceUrl, cancellationToken);
        }
        else
        {
            throw new BusinessException("No PDF content source available");
        }

        return ParsePdfBytes(pdfBytes, document);
    }

    private DocumentParseResult ParsePdfBytes(byte[] pdfBytes, RagDocument document)
    {
        var text = new StringBuilder();
        var warnings = new List<string>();
        int pageCount = 0;
        int tableCount = 0;
        var metadata = new DocumentMetadata();

        try
        {
            using var stream = new MemoryStream(pdfBytes);
            using var reader = new PdfReader(stream);
            using var pdfDoc = new PdfDocument(reader);

            pageCount = pdfDoc.GetNumberOfPages();

            // 提取元数据
            var docInfo = pdfDoc.GetDocumentInfo();
            if (docInfo != null)
            {
                metadata.Title = docInfo.GetTitle();
                metadata.Author = docInfo.GetAuthor();
                metadata.Subject = docInfo.GetSubject();
                metadata.CreatedDate = docInfo.GetCreationDate();
                metadata.ModifiedDate = docInfo.GetModificationDate();
            }

            // 逐页提取文本
            for (int i = 1; i <= pageCount; i++)
            {
                try
                {
                    var page = pdfDoc.GetPage(i);
                    var strategy = new SimpleTextExtractionStrategy();
                    var pageText = PdfTextExtractor.GetTextFromPage(page, strategy);

                    if (!string.IsNullOrWhiteSpace(pageText))
                    {
                        text.AppendLine($"=== Page {i} ===");
                        text.AppendLine(pageText);
                    }
                    else
                    {
                        warnings.Add($"Page {i} has no extractable text (may be scanned image)");
                    }
                }
                catch (Exception ex)
                {
                    logger.LogWarning(ex, "Failed to extract text from page {PageNumber}", i);
                    warnings.Add($"Failed to extract page {i}: {ex.Message}");
                }
            }
        }
        catch (Exception ex)
        {
            logger.LogError(ex, "Failed to parse PDF");
            throw new BusinessException($"Failed to parse PDF: {ex.Message}");
        }

        var content = text.ToString();
        if (string.IsNullOrWhiteSpace(content))
        {
            throw new BusinessException("No text extracted from PDF. It may be a scanned image requiring OCR.");
        }

        return new DocumentParseResult
        {
            Text = NormalizeText(content),
            TokenCount = EstimateTokens(content),
            ContentType = "application/pdf",
            Metadata = metadata,
            PageCount = pageCount,
            TableCount = tableCount,
            Warnings = warnings
        };
    }

    private string NormalizeText(string text)
    {
        // 移除多余空白
        text = System.Text.RegularExpressions.Regex.Replace(text, @"\s+", " ");
        // 修复常见编码问题
        text = text.Replace("ï¿½", "");
        return text.Trim();
    }

    private int EstimateTokens(string text)
    {
        return Math.Max(1, text.Length / 4);
    }
}
```

### 5.3 Word 文档解析器

```csharp
namespace KnowledgeBaseMod.Services;

using DocumentFormat.OpenXml.Packaging;
using DocumentFormat.OpenXml.Wordprocessing;

public class WordDocumentParser(
    ILogger<WordDocumentParser> logger
) : IDocumentParser
{
    public async Task<DocumentParseResult> ParseAsync(
        RagDocument document,
        string? rawContent,
        CancellationToken cancellationToken = default
    )
    {
        if (string.IsNullOrWhiteSpace(rawContent) && string.IsNullOrWhiteSpace(document.SourceUrl))
        {
            throw new BusinessException("Word document content must be provided");
        }

        byte[] docBytes;

        if (!string.IsNullOrWhiteSpace(rawContent))
        {
            try
            {
                docBytes = Convert.FromBase64String(rawContent);
            }
            catch
            {
                docBytes = Encoding.UTF8.GetBytes(rawContent);
            }
        }
        else
        {
            using var client = new HttpClient();
            docBytes = await client.GetByteArrayAsync(document.SourceUrl, cancellationToken);
        }

        return ParseWordBytes(docBytes, document);
    }

    private DocumentParseResult ParseWordBytes(byte[] docBytes, RagDocument document)
    {
        var text = new StringBuilder();
        var warnings = new List<string>();
        var metadata = new DocumentMetadata();
        int tableCount = 0;

        try
        {
            using var stream = new MemoryStream(docBytes);
            using var wordDoc = WordprocessingDocument.Open(stream, false);

            // 提取元数据
            var props = wordDoc.PackageProperties;
            if (props != null)
            {
                metadata.Title = props.Title;
                metadata.Author = props.Creator;
                metadata.Subject = props.Subject;
                metadata.CreatedDate = props.Created;
                metadata.ModifiedDate = props.Modified;
            }

            var body = wordDoc.MainDocumentPart?.Document?.Body;
            if (body == null)
            {
                throw new BusinessException("Word document has no body content");
            }

            // 提取段落
            foreach (var paragraph in body.Elements<Paragraph>())
            {
                var paraText = paragraph.InnerText;
                if (!string.IsNullOrWhiteSpace(paraText))
                {
                    text.AppendLine(paraText);
                }
            }

            // 提取表格内容
            foreach (var table in body.Elements<Table>())
            {
                tableCount++;
                text.AppendLine("\n=== Table ===");

                foreach (var row in table.Elements<TableRow>())
                {
                    var cells = row.Elements<TableCell>();
                    var rowContent = cells.Select(c => c.InnerText.Trim()).ToList();
                    text.AppendLine(string.Join(" | ", rowContent));
                }

                text.AppendLine();
            }
        }
        catch (Exception ex)
        {
            logger.LogError(ex, "Failed to parse Word document");
            throw new BusinessException($"Failed to parse Word document: {ex.Message}");
        }

        var content = text.ToString();
        if (string.IsNullOrWhiteSpace(content))
        {
            throw new BusinessException("No content extracted from Word document");
        }

        return new DocumentParseResult
        {
            Text = content.Trim(),
            TokenCount = EstimateTokens(content),
            ContentType = "application/vnd.openxmlformats-officedocument.wordprocessingml.document",
            Metadata = metadata,
            TableCount = tableCount,
            Warnings = warnings
        };
    }

    private int EstimateTokens(string text) => Math.Max(1, text.Length / 4);
}
```

```csharp
namespace KnowledgeBaseMod.Services;

using OfficeOpenXml;

public class ExcelDocumentParser(
    ILogger<ExcelDocumentParser> logger
) : IDocumentParser
{
    public async Task<DocumentParseResult> ParseAsync(
        RagDocument document,
        string? rawContent,
        CancellationToken cancellationToken = default
    )
    {
        byte[] excelBytes;

        if (!string.IsNullOrWhiteSpace(rawContent))
        {
            try
            {
                excelBytes = Convert.FromBase64String(rawContent);
            }
            catch
            {
                excelBytes = Encoding.UTF8.GetBytes(rawContent);
            }
        }
        else if (!string.IsNullOrWhiteSpace(document.SourceUrl))
        {
            using var client = new HttpClient();
            excelBytes = await client.GetByteArrayAsync(document.SourceUrl, cancellationToken);
        }
        else
        {
            throw new BusinessException("Excel content must be provided");
        }

        return ParseExcelBytes(excelBytes);
    }

    private DocumentParseResult ParseExcelBytes(byte[] excelBytes)
    {
        var text = new StringBuilder();
        var warnings = new List<string>();

        try
        {
            EPPlus.LicenseContext.CurrentLicenseContext = EPPlus.LicenseContext.NonCommercial;

            using var stream = new MemoryStream(excelBytes);
            using var package = new ExcelPackage(stream);

            foreach (var worksheet in package.Workbook.Worksheets)
            {
                text.AppendLine($"=== Sheet: {worksheet.Name} ===");
                text.AppendLine();

                var dimension = worksheet.Dimension;
                if (dimension == null)
                {
                    warnings.Add($"Sheet '{worksheet.Name}' is empty");
                    continue;
                }

                // 提取数据
                for (int row = dimension.Start.Row; row <= dimension.End.Row; row++)
                {
                    var rowData = new List<string>();

                    for (int col = dimension.Start.Column; col <= dimension.End.Column; col++)
                    {
                        var cell = worksheet.Cells[row, col];
                        var value = cell.Value?.ToString() ?? string.Empty;

                        if (!string.IsNullOrWhiteSpace(value))
                        {
                            rowData.Add(value);
                        }
                    }

                    if (rowData.Count > 0)
                    {
                        text.AppendLine(string.Join(" | ", rowData));
                    }
                }

                text.AppendLine();
            }
        }
        catch (Exception ex)
        {
            logger.LogError(ex, "Failed to parse Excel");
            throw new BusinessException($"Failed to parse Excel: {ex.Message}");
        }

        var content = text.ToString();
        if (string.IsNullOrWhiteSpace(content))
        {
            throw new BusinessException("No content extracted from Excel");
        }

        return new DocumentParseResult
        {
            Text = content.Trim(),
            TokenCount = EstimateTokens(content),
            ContentType = "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
            Warnings = warnings
        };
    }

    private int EstimateTokens(string text) => Math.Max(1, text.Length / 4);
}
```

### 5.5 PowerPoint 解析器

```csharp
namespace KnowledgeBaseMod.Services;

using DocumentFormat.OpenXml.Packaging;
using DocumentFormat.OpenXml.Presentation;
using DocumentFormat.OpenXml.Drawing;

public class PowerPointDocumentParser(
    ILogger<PowerPointDocumentParser> logger
) : IDocumentParser
{
    public async Task<DocumentParseResult> ParseAsync(
        RagDocument document,
        string? rawContent,
        CancellationToken cancellationToken = default
    )
    {
        byte[] pptBytes;

        if (!string.IsNullOrWhiteSpace(rawContent))
        {
            pptBytes = Convert.FromBase64String(rawContent);
        }
        else if (!string.IsNullOrWhiteSpace(document.SourceUrl))
        {
            using var client = new HttpClient();
            pptBytes = await client.GetByteArrayAsync(document.SourceUrl, cancellationToken);
        }
        else
        {
            throw new BusinessException("PowerPoint content required");
        }

        return ParsePowerPointBytes(pptBytes);
    }

    private DocumentParseResult ParsePowerPointBytes(byte[] pptBytes)
    {
        var text = new StringBuilder();

        try
        {
            using var stream = new MemoryStream(pptBytes);
            using var pptDoc = PresentationDocument.Open(stream, false);

            int slideCount = 0;

            foreach (var slidePart in pptDoc.PresentationPart!.SlideParts)
            {
                slideCount++;
                text.AppendLine($"=== Slide {slideCount} ===");

                var slide = slidePart.Slide;
                var shapes = slide.Descendants<Shape>();

                foreach (var shape in shapes)
                {
                    var textBody = shape.TextBody;
                    if (textBody != null)
                    {
                        foreach (var paragraph in textBody.Descendants<A.Paragraph>())
                        {
                            var paraText = string.Join("", paragraph.Descendants<A.Text>().Select(t => t.Text));
                            if (!string.IsNullOrWhiteSpace(paraText))
                            {
                                text.AppendLine(paraText);
                            }
                        }
                    }
                }

                text.AppendLine();
            }

            if (slideCount == 0)
            {
                throw new BusinessException("No slides found in PowerPoint");
            }
        }
        catch (Exception ex)
        {
            logger.LogError(ex, "Failed to parse PowerPoint");
            throw new BusinessException($"Failed to parse PowerPoint: {ex.Message}");
        }

        return new DocumentParseResult
        {
            Text = text.ToString().Trim(),
            TokenCount = EstimateTokens(text.ToString()),
            ContentType = "application/vnd.openxmlformats-officedocument.presentationml.presentation",
            PageCount = pptBytes.Length // 近似值
        };
    }

    private int EstimateTokens(string text) => Math.Max(1, text.Length / 4);
}
```

```csharp
[HttpPost("upload")]
public async Task<ActionResult<RagDocument>> UploadAsync(
    IFormFile file,
    [FromForm] Guid collectionId,
    CancellationToken cancellationToken
)
{
    if (file.Length == 0)
    {
        return BadRequest("File is empty");
    }

    const long maxFileSize = 50 * 1024 * 1024; // 50 MB
    if (file.Length > maxFileSize)
    {
        return BadRequest($"File too large (max {maxFileSize / 1024 / 1024}MB)");
    }

    // 读取文件内容
    using var stream = file.OpenReadStream();
    using var ms = new MemoryStream();
    await stream.CopyToAsync(ms, cancellationToken);
    var fileContent = ms.ToArray();

    // 调用 Manager 上传
    var dto = new RagDocumentAddDto
    {
        Title = Path.GetFileNameWithoutExtension(file.FileName),
        FileName = file.FileName,
        RagCollectionId = collectionId,
        ContentType = file.ContentType
    };

    var document = await _manager.UploadAsync(dto, fileContent, cancellationToken);
    return CreatedAtAction(nameof(GetAsync), new { id = document.Id }, document);
}
```
